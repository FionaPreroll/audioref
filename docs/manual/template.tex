% The LaTeX template of the user manual, for pandoc and LuaLaTeX, see utils/build-manual.sh and
% User manual in docs/DEVELOPMENT.md. It takes the variables title, product, subtitle, version, commit,
% lang (babel-lang), title-image, toc-title, tagline and license-note, and the body of the manual.
\documentclass[paper=a4,fontsize=11pt,oneside,open=any,parskip=half-,headings=normal,
  toc=chapterentrydotfill,DIV=12]{scrreprt}

\usepackage{fontspec}
\usepackage[$babel-lang$]{babel}
\usepackage{newunicodechar}

\newunicodechar{≥}{\ensuremath{\geq}}

% The plugin's accent, #BE03FD, darkened to read well on paper.
\definecolor{accent}{HTML}{7A0BA8}
\definecolor{accentlight}{HTML}{F5EBFA}
\definecolor{rule}{HTML}{BBBBBB}
\definecolor{muted}{HTML}{555555}

\usepackage{graphicx}
\makeatletter
% Pictures as wide as the text at most, and never taller than half a page.
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth\linewidth\else\Gin@nat@width\fi}
\def\maxheight{\ifdim\Gin@nat@height>0.5\textheight0.5\textheight\else\Gin@nat@height\fi}
\makeatother
\setkeys{Gin}{width=\maxwidth,height=\maxheight,keepaspectratio}
% pandoc 3.2 and later wrap pictures in \pandocbounded.
\providecommand{\pandocbounded}[1]{#1}
\usepackage{float}
\makeatletter
\def\fps@figure{!htbp}
\makeatother
% Let figures share pages with text rather than leave pages half empty.
\renewcommand{\topfraction}{0.85}
\renewcommand{\bottomfraction}{0.6}
\renewcommand{\textfraction}{0.12}
\renewcommand{\floatpagefraction}{0.75}
\usepackage[font={small,color=muted},labelfont={bf,color=accent},labelsep=period,
  skip=6pt]{caption}

% Tables as pandoc writes them, with the rules of booktabs.
\usepackage{longtable,booktabs,array,calc}
\setlength{\LTpre}{8pt}
\setlength{\LTpost}{8pt}
\renewcommand{\arraystretch}{1.25}
\AtBeginEnvironment{longtable}{\small}

\usepackage{needspace}
\usepackage{enumitem}
\setlist{itemsep=2pt,topsep=4pt}
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

\usepackage{fvextra}

\fvset{fontsize=\small,breaklines=true}
$if(highlighting-macros)$
$highlighting-macros$
$endif$

% Notes, tips and warnings: blocks with a bar in the accent, see manual.lua.

  before skip=10pt,after skip=10pt,fontupper=\small,
  title={#1},attach title to upper={\par\smallskip},
  fonttitle=\sffamily\bfseries\small,coltitle=accent}
% Code blocks: on a light grey ground.
\definecolor{shadecolor}{HTML}{F4F4F4}
\makeatletter
\@ifundefined{Shaded}{\newenvironment{Shaded}{}{}}{}
\makeatother

  {\end{tcolorbox}}

\setkomafont{section}{\Large}
\setkomafont{subsection}{\large}

\RedeclareSectionCommand[beforeskip=0pt,afterskip=1.6\baselineskip]{chapter}
\RedeclareSectionCommand[beforeskip=-1.4\baselineskip,afterskip=0.6\baselineskip]{section}

\renewcommand*{\chapterformat}{%
  {\fontsize{40}{40}\selectfont\color{accent!45}\thechapter}\enskip}
\renewcommand*{\chapterlinesformat}[3]{%
  \parbox[b]{\linewidth}{\raggedright #2#3}\par\nobreak\vspace{4pt}%
  {\color{accent}\rule{\linewidth}{0.8pt}}}
\setcounter{secnumdepth}{2}
\setcounter{tocdepth}{1}
\setkomafont{chapterentry}{\bfseries\color{accent}}
